{{- /* Usage: {{< video name="AT-ACT-V01-Rotation-X" caption="…" >}} Optional: dir="/path/" (default: page path without language prefix, files in static/), muted="true", autoplay="true" (implies muted, behaves like a GIF), poster="false" */ -}} {{- $name := .Get "name" -}} {{- $dir := .Get "dir" | default (replaceRE `^/[a-z]{2}/` "/" .Page.RelPermalink) -}} {{- $autoplay := eq (.Get "autoplay") "true" -}} {{- $muted := or $autoplay (eq (.Get "muted") "true") -}} {{- /* Fail the build instead of serving a silent 404 when the video file is missing */ -}} {{- $file := printf "static%s%s.mp4" $dir $name -}} {{- if not (fileExists $file) -}} {{- errorf "video shortcode: file not found: %s (page %s)" $file .Page.File.Path -}} {{- end -}} {{- if and (ne (.Get "poster") "false") (not (fileExists (printf "static%s%s-poster.png" $dir $name))) -}} {{- warnf "video shortcode: poster not found: static%s%s-poster.png (page %s)" $dir $name .Page.File.Path -}} {{- end -}}
