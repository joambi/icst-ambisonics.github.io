{{- /* Inlines an SVG from the page bundle so it inherits text colour (dark mode). Usage: {{< inline-svg src="anim-startwert" >}} -> loads anim-startwert-.svg if present, else anim-startwert.svg */ -}} {{- $src := .Get "src" -}} {{- $lang := .Page.Language.Lang -}} {{- $res := or (.Page.Resources.Get (printf "%s-%s.svg" $src $lang)) (.Page.Resources.Get (printf "%s.svg" $src)) -}} {{- with $res -}} {{- else -}} {{- errorf "inline-svg: %s not found in %s" $src $.Page.File.Path -}} {{- end -}}
